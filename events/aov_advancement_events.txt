# Lichdom level 3: choose the prisoner to sacrifice (the three oldest)
aov_advancement.40 = {
	type = character_event
	title = aov_advancement.40.t
	desc = aov_advancement.40.desc
	theme = dread
	left_portrait = root
	right_portrait = scope:aov_sacrifice_1

	# A position past the end of the list caps at the last prisoner, so later positions need that many prisoners
	immediate = {
		ordered_prisoner = {
			order_by = age
			position = 0
			save_scope_as = aov_sacrifice_1
		}
		if = {
			limit = { any_prisoner = { count >= 2 } }
			ordered_prisoner = {
				order_by = age
				position = 1
				save_scope_as = aov_sacrifice_2
			}
		}
		if = {
			limit = { any_prisoner = { count >= 3 } }
			ordered_prisoner = {
				order_by = age
				position = 2
				save_scope_as = aov_sacrifice_3
			}
		}
	}

	option = {
		name = aov_advancement.40.a
		trigger = { exists = scope:aov_sacrifice_1 }
		aov_adv_lich_ritual = { VICTIM = scope:aov_sacrifice_1 }
	}
	option = {
		name = aov_advancement.40.a2
		trigger = { exists = scope:aov_sacrifice_2 }
		aov_adv_lich_ritual = { VICTIM = scope:aov_sacrifice_2 }
	}
	option = {
		name = aov_advancement.40.a3
		trigger = { exists = scope:aov_sacrifice_3 }
		aov_adv_lich_ritual = { VICTIM = scope:aov_sacrifice_3 }
	}

	# The prisoners are gone: the ritual waits, ready, for one (nothing is paid again)
	option = {
		name = aov_advancement.40.b
		trigger = { NOT = { exists = scope:aov_sacrifice_1 } }
		aov_adv_lichdom_await_sacrifice = yes
	}
}
