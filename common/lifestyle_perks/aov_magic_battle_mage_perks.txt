# Spark of Battle
aov_spark_of_battle_perk = {
	lifestyle = magic_lifestyle
	tree = aov_battle_mage
	position = { 0 0 }
	icon = node_magic

	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				OR = {
					aov_is_studying = { SCHOOL = evocation }
					aov_is_studying = { SCHOOL = conjuration }
				}
			}
			add = 1989
		}
		if = {
			limit = {
				NOT = { has_variable = aov_studying }
				OR = {
					highest_skill = martial
					highest_skill = prowess
				}
			}
			add = 1000
		}
		if = {
			limit = {
				has_focus = magic_duelist_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = aov_summoners_circle_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		prowess = 2
	}

	effect = {
		custom_description_no_bullet = { text = aov_spark_of_battle_perk_effect }
		aov_magic_sync_mastery = yes
	}
}
