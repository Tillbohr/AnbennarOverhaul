# Artificery research popup and Inventions tab panel. Per-invention sGUIs are generated: aov_invention_sgui.txt.

aov_research_can_start_sgui = {
	scope = character
	is_valid = { aov_artificery_can_start_research = yes }
}

aov_research_running_sgui = {
	scope = character
	is_shown = { aov_artificery_is_researching = yes }
}

aov_research_has_target_sgui = {
	scope = character
	is_shown = { has_variable = aov_research_target }
}

aov_research_has_category_sgui = {
	scope = character
	is_shown = {
		has_variable = aov_research_category
		NOT = { has_variable = aov_research_target }
	}
}

aov_research_brillites_sgui = {
	scope = character
	is_valid = { aov_artificery_can_start_research = yes }
	effect = { aov_artificery_start_brillites_research = yes }
}

aov_research_mechanists_economic_sgui = {
	scope = character
	is_valid = {
		aov_artificery_can_start_research = yes
		custom_tooltip = { text = aov_artificery_category_discoverable_tt aov_inventions_economic_discoverable = yes }
	}
	effect = { aov_artificery_start_mechanists_research = { CATEGORY = economic } }
}

# Technomancers: viewing a category rolls its offers once per research cycle
aov_research_technomancers_economic_sgui = {
	scope = character
	is_valid = {
		aov_artificery_can_start_research = yes
		custom_tooltip = { text = aov_artificery_category_discoverable_tt aov_inventions_economic_discoverable = yes }
	}
	effect = {
		if = {
			limit = { NOT = { has_variable = aov_tech_offers_rolled_economic } }
			aov_inventions_roll_offers_economic = yes
		}
	}
}
