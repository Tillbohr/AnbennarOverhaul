magic_duelist_focus = {
	lifestyle = magic_lifestyle

	modifier = {
		prowess = 3
		martial = 1
	}

	desc = {
		desc = magic_duelist_focus_desc
		desc = line_break
		desc = magic_duelist_focus_study_desc
	}

	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				OR = {
					aov_is_studying = { SCHOOL = evocation }
					aov_is_studying = { SCHOOL = conjuration }
				}
			}
			add = 1989
		}
		if = {
			limit = {
				NOT = { has_variable = aov_studying }
				OR = {
					highest_skill = martial
					highest_skill = prowess
				}
			}
			add = 1000
		}
		if = {
			limit = {
				has_trait = brave
			}
			multiply = 5
		}
	}
}
