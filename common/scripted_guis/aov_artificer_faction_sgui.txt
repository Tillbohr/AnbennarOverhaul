# Anbennar Overhaul: artificer factions in the Factions tab (gui/aov_window_artificery.gui)

aov_make_amends_brillites_sgui = {
	scope = character
	is_valid = {
		custom_tooltip = {
			text = aov_make_amends_below_40_tt
			aov_influence_brillites < 40
		}
		custom_tooltip = {
			text = aov_make_amends_gold_tt
			gold >= 100
		}
	}
	effect = { aov_faction_make_amends = { FACTION = brillites } }
}

aov_faction_brillites_election_sgui = {
	scope = character
	is_shown = { exists = global_var:aov_election_open_brillites }
}
