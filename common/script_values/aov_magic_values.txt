# Anbennar Overhaul: magic schools. Character scope (the mage).

aov_affinity_level = {
	value = 0
	if = { limit = { has_trait = magical_affinity_1 } value = 1 }
	if = { limit = { has_trait = magical_affinity_2 } value = 2 }
	if = { limit = { has_trait = magical_affinity_3 } value = 3 }
}

aov_mana = {
	value = 0
	if = {
		limit = { has_variable = aov_mana }
		value = var:aov_mana
	}
}

# 100 + 50 per affinity level
aov_mana_max = {
	value = 100
	add = {
		value = aov_affinity_level
		multiply = 50
	}
}

# Per month: 5 + Learning / 2 + 2 per affinity level (applied quarterly x 3)
aov_mana_refill_month = {
	value = 5
	add = {
		value = learning
		divide = 2
	}
	add = {
		value = aov_affinity_level
		multiply = 2
	}
}

# Study per month: Learning + 5 per affinity level (applied quarterly x 3)
aov_study_gain_month = {
	value = learning
	add = {
		value = aov_affinity_level
		multiply = 5
	}
}

aov_school_abjuration_progress = {
	value = 0
	if = {
		limit = { has_variable = aov_school_abjuration_progress }
		value = var:aov_school_abjuration_progress
	}
}

aov_school_abjuration_level = {
	value = 0
	if = { limit = { aov_school_abjuration_progress >= 300 } value = 1 }
	if = { limit = { aov_school_abjuration_progress >= 800 } value = 2 }
	if = { limit = { aov_school_abjuration_progress >= 2000 } value = 3 }
}

# Progress needed for the next level (2000 once at level 3)
aov_school_abjuration_next = {
	value = 300
	if = { limit = { aov_school_abjuration_level >= 1 } value = 800 }
	if = { limit = { aov_school_abjuration_level >= 2 } value = 2000 }
}
