# Summon Elementals ends: the conjured army returns to its plane (army may already be gone with its war)
aov_magic.10 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { exists = var:aov_elemental_army }
			var:aov_elemental_army = { deplete_army_by_percentage = 1 }
		}
		remove_variable = aov_elemental_army
	}
}
