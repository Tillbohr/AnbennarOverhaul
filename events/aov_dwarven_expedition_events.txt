namespace = aov_expedition

# Anbennar Overhaul: events of the Dwarven Expedition (common/activities/activity_types/aov_dwarven_expedition.txt),
# on the pattern of Realms in Exile's mining events. scope:activity is the expedition throughout.
#	.1		setting out (the host, from aov_expedition_start_events)
#	.101-.106	pulses (a human attendee, from aov_expedition_random_pulse at months 2, 4 and 6); their options
#			move the success chance by aov_expedition_success_minor/medium/major (3/5/10), through
#			aov_expedition_change_success_effect, with a tooltip naming the step
#	.900-.902	outcomes (the host, AI or player, from aov_expedition_outcome at month 7): Mining success,
#			Mining failure, Archaeology
# Backgrounds: the gate when setting out; in the deep, the halls in a hold or on cavern / dwarven road terrain, else the
# cave (common/event_backgrounds/aov_dwarven_expedition_backgrounds.txt).

##################################################
# Setting out
##################################################

aov_expedition.1 = {
	type = activity_event
	title = aov_expedition.1.t
	theme = stewardship

	trigger = {
		exists = scope:activity
		scope:activity = { has_activity_type = activity_dwarven_expedition }
		scope:activity.activity_host = root
	}

	desc = aov_expedition.1.desc

	override_background = { reference = aov_expedition_gate }

	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		trigger = { exists = scope:aov_expedition_companion }
		character = scope:aov_expedition_companion
		animation = happiness
	}

	immediate = {
		scope:activity = {
			random_attending_character = {
				limit = { NOT = { this = root } }
				save_scope_as = aov_expedition_companion
			}
		}
	}

	option = { # a speech before the gate
		name = aov_expedition.1.a
		custom_tooltip = aov_expedition_success_gain_minor_tt
		scope:activity = { aov_expedition_change_success_effect = { AMOUNT = aov_expedition_success_minor_value } }
		add_prestige = miniscule_prestige_gain
		ai_chance = {
			base = 50
		}
	}

	option = { # the best lamps, rope and pit props
		name = aov_expedition.1.b
		trigger = {
			short_term_gold >= minor_gold_value
		}
		show_as_unavailable = { always = yes }
		remove_short_term_gold = minor_gold_value
		custom_tooltip = aov_expedition_success_gain_medium_tt
		scope:activity = { aov_expedition_change_success_effect = { AMOUNT = aov_expedition_success_medium_value } }
		stress_impact = {
			greedy = minor_stress_impact_gain
		}
		ai_chance = {
			base = 50
			modifier = {
				has_trait = greedy
				factor = 0.2
			}
		}
	}

	option = { # the oldest delver leads the way
		name = aov_expedition.1.c
		random_list = {
			50 = {
				custom_tooltip = aov_expedition_success_gain_medium_tt
				scope:activity = { aov_expedition_change_success_effect = { AMOUNT = aov_expedition_success_medium_value } }
			}
			50 = {
				custom_tooltip = aov_expedition_success_loss_minor_tt
				scope:activity = { aov_expedition_change_success_effect = { AMOUNT = aov_expedition_success_minor_loss_value } }
			}
		}
		ai_chance = {
			base = 25
		}
	}
}
