# Anbennar Overhaul: race relations (docs/systems/race-relations.md). A Vampire or Werewolf hides what they are; once
# the secret is exposed they count as monstrous for good (aov_race_relations_expose_effect).

secret_aov_vampire = {
	category = religious

	is_valid = {
		scope:secret_owner = {
			has_trait = vampire
			NOT = { has_character_flag = aov_exposed_monster }
		}
	}

	is_shunned = {
		always = yes
	}

	is_criminal = {
		always = no
	}

	on_expose = {
		save_scope_as = secret
		secret_exposed_notification_effect = yes
		scope:secret_owner = { aov_race_relations_expose_effect = yes }
	}
}
