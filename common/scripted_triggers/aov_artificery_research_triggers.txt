# Artificery inventions: character-scope triggers

aov_artificery_is_researching = {
	has_variable = aov_research_sponsor
}

aov_artificery_can_start_research = {
	can_use_artificery_trigger = yes
	custom_tooltip = {
		text = aov_artificery_not_researching_tt
		NOT = { aov_artificery_is_researching = yes }
	}
	custom_tooltip = {
		text = aov_artificery_research_gold_tt
		gold >= aov_research_cost
	}
	custom_tooltip = {
		text = aov_artificery_any_discoverable_tt
		aov_inventions_any_discoverable = yes
	}
}

aov_artificery_has_free_slot = {
	custom_tooltip = {
		text = aov_artificery_free_slot_tt
		aov_invention_slots_free > 0
	}
}

# Has anything worth suspending or inheriting
aov_artificery_has_state = {
	OR = {
		aov_artificery_is_researching = yes
		aov_inventions_any_discovered = yes
	}
}
