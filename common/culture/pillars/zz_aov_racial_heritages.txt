# Anbennar Overhaul: racial heritages. Overrides Anbennar's heritage_elven, heritage_dwarven, heritage_gnomish and
# heritage_kobold (common/culture/pillars/anb_heritage.txt): type, is_shown and audio_parameter are copied unchanged;
# the parameters and character_modifier are new. See docs/systems/racial-armies.md.
#
# Levy count x levy quality equals a human's (elves 0.25 x 4, dwarves 0.6 x 1.7, gnomes 0.5 x 2, kobolds 1.5 x 0.7).
# The aov_racial_levy_* parameter marks the race for the per-holding levy correction
# (common/scripted_effects/aov_racial_levy_effects.txt), which keeps that share fixed as levy bonuses are added.

# Elves

heritage_elven = {
	type = heritage
	is_shown = {
		heritage_is_shown_trigger = {
			HERITAGE = heritage_elven
		}
	}
	parameters = {
		aov_racial_levy_elven = yes
	}
	character_modifier = {
		levy_size = -0.75
		levy_attack = 30
		levy_toughness = 30
		domain_limit = -1
	}
	audio_parameter = european
}
