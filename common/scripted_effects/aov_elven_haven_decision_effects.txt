# Anbennar Overhaul: the Raise an Elven Haven decision's option effect (docs/systems/rie-elves.md).

# Character scope (the ruler); scope:$CHOICE$ is the chosen barony title. Pays 100 gold per Haven level and 100
# prestige, converts the barony and closes the decision for 5 years (cancelling the picker does not).
aov_raise_elven_haven_effect = {
	add_character_flag = { flag = aov_raised_elven_haven years = 5 }
	scope:$CHOICE$.title_province = {
		save_scope_value_as = { name = aov_haven_cost value = aov_elven_haven_conversion_cost }
	}
	remove_short_term_gold = scope:aov_haven_cost
	add_prestige = -100
	scope:$CHOICE$.title_province = { aov_convert_to_elven_haven_effect = yes }
}
