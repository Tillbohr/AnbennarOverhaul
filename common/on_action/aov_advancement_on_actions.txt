# Anbennar Overhaul: magic advancements on_actions, appended to vanilla on_actions.

quarterly_playable_pulse = {
	on_actions = { aov_adv_quarterly_pulse }
}

# Root = character. Three months of progress on the advancement in progress (aov_advancement_effects.txt, generated).
# A committed character who lost their affinity still ticks, so aov_adv_quarterly abandons the commitment
aov_adv_quarterly_pulse = {
	trigger = {
		OR = {
			aov_is_mage = yes
			aov_adv_is_committed = yes
		}
	}
	effect = {
		aov_adv_quarterly = yes
		aov_adv_lichdom_check = yes
		aov_adv_artifact_check = yes
	}
}

yearly_playable_pulse = {
	on_actions = { aov_adv_wards_yearly aov_adv_ai_yearly }
}

# Root = character. Magical Fortress: re-lay the wards (aov_advancement_level_effects.txt)
aov_adv_wards_yearly = {
	trigger = {
		aov_adv_eligible = yes
		aov_adv_magical_fortress_level >= 1
	}
	effect = {
		aov_adv_wards_refresh = yes
	}
}

on_imprison = {
	on_actions = { aov_adv_body_double }
}

# Root = the imprisoned character. Mirror Image level 2: a body double may be taken instead; the release waits a day
# so vanilla's on_imprison effects finish first (event aov_advancement.50)
aov_adv_body_double = {
	trigger = {
		aov_adv_mirror_image_level >= 2
	}
	effect = {
		random = {
			chance = aov_adv_body_double_chance
			trigger_event = { id = aov_advancement.50 days = 1 }
		}
	}
}

# Root = character. An AI ruling mage who is not working on an advancement may start one (aov_adv_ai_pick, generated)
aov_adv_ai_yearly = {
	trigger = {
		is_ai = yes
		aov_adv_eligible = yes
		NOT = { aov_adv_is_committed = yes }
	}
	effect = {
		random = {
			chance = aov_adv_ai_chance
			aov_adv_ai_pick = yes
		}
	}
}

on_title_lost = {
	on_actions = { aov_adv_on_title_lost }
}

# Root = the old holder. A mage who loses their last title no longer gets quarterly_playable_pulse, so the commitment
# is checked here, a day later once the title has changed hands (event aov_advancement.60)
aov_adv_on_title_lost = {
	trigger = {
		aov_adv_is_committed = yes
	}
	effect = {
		trigger_event = { id = aov_advancement.60 days = 1 }
	}
}
