# Anbennar Overhaul: magic schools. Character scope (the mage).

aov_affinity_level = {
	value = 0
	if = { limit = { has_trait = magical_affinity_1 } value = 1 }
	if = { limit = { has_trait = magical_affinity_2 } value = 2 }
	if = { limit = { has_trait = magical_affinity_3 } value = 3 }
}

# Full until first set: a new mage starts with full mana
aov_mana = {
	value = aov_mana_max
	if = {
		limit = { has_variable = aov_mana }
		value = var:aov_mana
	}
}

# 100 + 50 per affinity level + perk/trait bonus (aov_magic_lifestyle_values.txt)
aov_mana_max = {
	value = 100
	add = {
		value = aov_affinity_level
		multiply = 50
	}
	add = aov_mana_max_bonus
}

# Per month: (5 + Learning / 2 + 2 per affinity level) x aov_mana_refill_mult (applied quarterly x 3)
aov_mana_refill_month = {
	value = 5
	add = {
		value = learning
		divide = 2
	}
	add = {
		value = aov_affinity_level
		multiply = 2
	}
	multiply = aov_mana_refill_mult
}

# Study per month: Learning + 5 per affinity level, before the per-school multiplier (applied quarterly x 3)
aov_study_gain_month = {
	value = learning
	add = {
		value = aov_affinity_level
		multiply = 5
	}
}

# Abjuration study per month, with perks, focus and trait
aov_study_gain_abjuration_month = {
	value = aov_study_gain_month
	multiply = aov_study_mult_abjuration

# Study per month in the school being studied (0 if none)
aov_study_gain_current_month = {
	value = 0
	if = { limit = { aov_is_studying = { SCHOOL = abjuration } } value = aov_study_gain_abjuration_month }
	if = { limit = { aov_is_studying = { SCHOOL = conjuration } } value = aov_study_gain_conjuration_month }
	if = { limit = { aov_is_studying = { SCHOOL = divination } } value = aov_study_gain_divination_month }
	if = { limit = { aov_is_studying = { SCHOOL = enchantment } } value = aov_study_gain_enchantment_month }
	if = { limit = { aov_is_studying = { SCHOOL = evocation } } value = aov_study_gain_evocation_month }
	if = { limit = { aov_is_studying = { SCHOOL = illusion } } value = aov_study_gain_illusion_month }
	if = { limit = { aov_is_studying = { SCHOOL = necromancy } } value = aov_study_gain_necromancy_month }
	if = { limit = { aov_is_studying = { SCHOOL = transmutation } } value = aov_study_gain_transmutation_month }
}

aov_school_abjuration_progress = {
	value = 0
	if = {
		limit = { has_variable = aov_school_abjuration_progress }
		value = var:aov_school_abjuration_progress
	}
}

aov_school_abjuration_level = {
	value = 0
	if = { limit = { aov_school_abjuration_progress >= 300 } value = 1 }
	if = { limit = { aov_school_abjuration_progress >= 800 } value = 2 }
	if = { limit = { aov_school_abjuration_progress >= 2000 } value = 3 }
}

# Progress needed for the next level (2000 once at level 3)

	value = 300
	if = { limit = { aov_school_abjuration_level >= 1 } value = 800 }
	if = { limit = { aov_school_abjuration_level >= 2 } value = 2000 }
}
