# Anbennar Overhaul: Dwarovar tribes (docs/systems/dwarovar-tribes.md). A tribe's titles come from the pool of blank
# titles, because the normal create-title route needs de jure land the tribe has no right to.

aov_form_tribal_duchy_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = major
	sort_order = 92
	ai_check_interval = 60

	desc = aov_form_tribal_duchy_decision_desc
	selection_tooltip = aov_form_tribal_duchy_decision_tooltip
	confirm_text = aov_form_tribal_duchy_decision_confirm

	is_shown = {
		is_landed = yes
		government_has_flag = government_is_dwarovar_tribe
		highest_held_title_tier = tier_county
	}

	is_valid = {
		sub_realm_size >= aov_form_duchy_counties
		custom_description = {
			text = aov_tribe_no_free_title
			aov_tribe_has_free_title_trigger = { TIER = duchy }
		}
		custom_description = {
			text = aov_tribe_would_rival_liege
			OR = {
				is_independent_ruler = yes
				liege ?= { highest_held_title_tier > tier_duchy }
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
	}

	cost = {
		prestige = aov_form_duchy_prestige
	}

	effect = {
		aov_tribe_take_free_title_effect = { TIER = duchy }
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:aov_new_title = {
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
	}

	ai_potential = {
		is_at_war = no
	}

	# The AI waits until it has as much prestige again in reserve, so a whole region does not rank up in the first year.
	ai_will_do = {
		base = 50
		modifier = {
			factor = 0
			prestige < {
				value = aov_form_duchy_prestige
				multiply = 2
			}
		}
	}
}

# An orc vassal of an independent warband challenges their ruler for the throne (events/aov_tribes_duel_events.txt).
aov_challenge_warlord_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}
	decision_group_type = major
	sort_order = 95
	ai_check_interval = 36

	desc = aov_challenge_warlord_decision_desc
	selection_tooltip = aov_challenge_warlord_decision_tooltip
	confirm_text = aov_challenge_warlord_decision_confirm

	cooldown = { years = 5 }

	is_shown = {
		has_trait = race_orc
		is_landed = yes
		is_imprisoned = no
		can_start_single_combat_trigger = yes
		liege ?= {
			this != root
			is_independent_ruler = yes
			government_has_flag = government_is_orcish_warband
		}
	}

	is_valid_showing_failures_only = {
		is_adult = yes
		is_at_war = no
		can_be_combatant_based_on_gender_trigger = { ARMY_OWNER = liege }
		liege = {
			is_available = yes
			is_adult = yes
			is_imprisoned = no
			is_at_war = no
			can_start_single_combat_trigger = yes
		}
	}

	effect = {
		save_scope_as = challenger
		liege = {
			save_scope_as = ruler
			trigger_event = aov_tribe_duel.1
		}
		custom_tooltip = aov_challenge_warlord_effect_tooltip
	}

	ai_potential = {
		is_adult = yes
		OR = {
			has_trait = ambitious
			prowess >= {
				value = liege.prowess
				add = 6
			}
		}
	}

	# Only ever from strength, more so the further ahead.
	ai_will_do = {
		base = 0
		modifier = {
			add = 25
			has_trait = ambitious
		}
		modifier = {
			add = 25
			prowess >= {
				value = liege.prowess
				add = 6
			}
		}
		modifier = {
			add = 25
			prowess >= {
				value = liege.prowess
				add = 12
			}
		}
		modifier = {
			factor = 0
			prowess < liege.prowess
		}
	}
}
