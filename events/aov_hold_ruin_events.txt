# A hold was conquered and fell into ruin (scope:hold_province, scope:hold_title). Sent to the new and the old holder.
aov_hold_ruin.1 = {
	type = character_event
	title = aov_hold_ruin.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:hold_province = { barony.holder = root } }
				desc = aov_hold_ruin.1.desc.new
			}
			desc = aov_hold_ruin.1.desc.old
		}
	}
	theme = war
	override_background = { reference = tgp_ruined_holding }
	left_portrait = root

	option = {
		name = aov_hold_ruin.1.a
	}
}

# The restoration is done (scope:hold_province, scope:hold_title). Resolved on the hold's current holder:
# a dwarf completes the restoration, anyone else cancels it.
aov_hold_ruin.3 = {
	type = character_event
	title = aov_hold_ruin.3.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:restored = yes }
				desc = aov_hold_ruin.3.desc.restored
			}
			desc = aov_hold_ruin.3.desc.lost
		}
	}
	theme = realm
	override_background = { reference = wilderness_mountains }
	left_portrait = root

	trigger = {
		scope:hold_province = {
			has_variable = aov_restore_pending
			NOT = { has_variable = aov_restore_due }
			barony.holder = root
		}
	}

	immediate = {
		scope:hold_province = {
			remove_variable = aov_restore_pending
			remove_variable = aov_restorer
			remove_province_modifier = aov_restoring_hold
		}
		if = {
			limit = { is_dwarvish_culture = yes }
			save_scope_value_as = { name = restored value = yes }
			scope:hold_province = {
				remove_variable = aov_hold_ruined
				county = {
					remove_county_modifier = aov_ruined_hold
				}
			}
		}
		else = {
			save_scope_value_as = { name = restored value = no }
		}
	}

	option = {
		name = aov_hold_ruin.3.a
	}
}

# Hidden: a restoration passed to a new holder. Waits yearly until the restoration is due, then fires .3.
aov_hold_ruin.4 = {
	hidden = yes

	trigger = {
		scope:hold_province = {
			has_variable = aov_restore_pending
			barony.holder = root
		}
	}

	immediate = {
		if = {
			limit = {
				scope:hold_province = { has_variable = aov_restore_due }
			}
			trigger_event = { id = aov_hold_ruin.4 years = 1 }
		}
		else = {
			trigger_event = { id = aov_hold_ruin.3 days = 1 }
		}
	}
}
