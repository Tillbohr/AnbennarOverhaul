# Anbennar Overhaul: dwarven hold on_actions, appended to vanilla on_actions.

on_game_start = {
	on_actions = { aov_hold_game_start }
}

# Converts the holds, then places the ore deposits present at game start (Rubyhold, Verkal Gulan, Anvilwright).
aov_hold_game_start = {
	effect = {
		aov_hold_setup_effect = yes
		aov_rie_deposit_setup_effect = yes
	}
}

quarterly_playable_pulse = {
	on_actions = { aov_hold_dig_quarterly }
}

# Root = a playable ruler. Each running dig at a hold the ruler holds advances 3 months (the finest global pulse is
# quarterly). A dig pauses while its hold is occupied or its holder is not a dwarf.
aov_hold_dig_quarterly = {
	trigger = {
		is_dwarvish_culture = yes
	}
	effect = {
		every_held_title = {
			limit = {
				tier = tier_barony
				title_province ?= {
					aov_is_hold = yes
					aov_hold_dig_running = yes
					is_occupied = no
				}
			}
			title_province = {
				aov_hold_add_dig_progress_effect = { MONTHS = 3 }
			}
		}
	}
}
