# Random dig events, one per pulse, player campaigns only (root = the host, activity = involved_activity;
# progress changes are capped at 20 either way). Chosen by common/on_action/aov_hold_dig_on_actions.txt.
aov_hold_dig.1 = {
	type = character_event
	title = aov_hold_dig.1.t
	desc = aov_hold_dig.1.desc
	theme = stewardship
	override_background = { reference = fp3_cave }
	left_portrait = root

	trigger = {
		exists = involved_activity
		involved_activity = { has_activity_type = activity_hold_dig }
		involved_activity = { has_variable = aov_campaign_progress }
	}

	immediate = {
		involved_activity = { save_scope_as = activity }
		scope:activity.var:aov_dig_hold = { save_scope_as = hold }
	}

	option = {
		name = aov_hold_dig.1.a
		scope:activity = { change_variable = { name = aov_campaign_progress add = 10 } }
	}

	option = {
		name = aov_hold_dig.1.b
		trigger = { gold >= 60 }
		show_as_unavailable = { always = yes }
		remove_short_term_gold = 60
		scope:activity = { change_variable = { name = aov_campaign_progress add = 16 } }
	}

	option = {
		name = aov_hold_dig.1.c
		scope:activity = { change_variable = { name = aov_campaign_progress add = 3 } }
		add_prestige = 25
	}
}

# Breakthrough at a hold: the Delving Campaign raised it one level (scope:hold_province, scope:hold_title).
aov_hold_dig.100 = {
	type = character_event
	title = aov_hold_dig.100.t
	desc = {
		desc = aov_hold_dig.100.desc
		first_valid = {
			triggered_desc = {
				trigger = { scope:aov_discovery = flag:gems }
				desc = aov_hold_dig.100.desc.gems
			}
			triggered_desc = {
				trigger = { scope:aov_discovery = flag:vein }
				desc = aov_hold_dig.100.desc.vein
			}
			desc = aov_hold_dig.100.desc.lore
		}
	}
	theme = stewardship
	override_background = { reference = fp3_cave }
	left_portrait = root

	immediate = {
		random_list = {
			1 = { save_scope_value_as = { name = aov_discovery value = flag:gems } }
			1 = { save_scope_value_as = { name = aov_discovery value = flag:vein } }
			1 = { save_scope_value_as = { name = aov_discovery value = flag:lore } }
		}
	}

	option = {
		name = aov_hold_dig.100.a
		if = {
			limit = { scope:aov_discovery = flag:gems }
			add_gold = {
				value = scope:hold_province.aov_hold_level
				multiply = 25
			}
		}
		else_if = {
			limit = { scope:aov_discovery = flag:vein }
			scope:hold_province.county = {
				add_county_modifier = {
					modifier = aov_hold_rich_vein
					years = 10
				}
			}
		}
		else = {
			add_learning_lifestyle_xp = 500
		}
	}
}
