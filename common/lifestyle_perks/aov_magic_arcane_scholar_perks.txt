# Diviner's Eye
aov_diviners_eye_perk = {
	lifestyle = magic_lifestyle
	tree = aov_arcane_scholar
	position = { 0 0 }
	icon = node_magic

	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				OR = {
					aov_is_studying = { SCHOOL = divination }
					aov_is_studying = { SCHOOL = abjuration }
					aov_is_studying = { SCHOOL = transmutation }
				}
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = magic_arcane_study_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = aov_warding_glyphs_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		learning = 1
	}

	effect = {
		custom_description_no_bullet = { text = aov_diviners_eye_perk_effect }
		aov_magic_sync_mastery = yes
	}
}
