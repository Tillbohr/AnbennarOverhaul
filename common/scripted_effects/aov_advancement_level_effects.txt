# Wards (common/modifiers/aov_advancement_modifiers.txt) on the counties the mage holds: the capital at level 1, the
# whole domain from level 2. Re-laid yearly (aov_adv_wards_yearly); they lapse after aov_adv_ward_days when the mage
# dies, loses the county or stops being a ruling mage.
aov_adv_wards_refresh = {
	every_held_county = {
		remove_county_modifier = aov_ward_1
		remove_county_modifier = aov_ward_3
		remove_county_modifier = aov_ward_capital
	}
	if = {
		limit = { aov_adv_magical_fortress_level = 1 }
		capital_county = {
			add_county_modifier = { modifier = aov_ward_1 days = aov_adv_ward_days }
		}
	}
	else_if = {
		limit = { aov_adv_magical_fortress_level = 2 }
		every_held_county = {
			add_county_modifier = { modifier = aov_ward_1 days = aov_adv_ward_days }
		}
		capital_county = {
			add_county_modifier = { modifier = aov_ward_capital days = aov_adv_ward_days }
		}
	}
	else_if = {
		limit = { aov_adv_magical_fortress_level >= 3 }
		every_held_county = {
			add_county_modifier = { modifier = aov_ward_3 days = aov_adv_ward_days }
		}
		capital_county = {
			add_county_modifier = { modifier = aov_ward_capital days = aov_adv_ward_days }
		}
	}
}

# Divination: Orb of Omniscience
##########

# The Orb (court artifact, var:aov_adv_orb). Its modifiers stay with the artifact; the spell boosts
# (aov_scry_reveal, aov_adv_orb_spell_years_mult) need a caster with the level.
aov_adv_orb_of_omniscience_1_effect = {
	add_magic_lifestyle_xp = 100
	save_scope_as = owner
	set_artifact_rarity_famed = yes
	create_artifact = {
		name = aov_orb_of_omniscience_name
		description = aov_orb_of_omniscience_desc
		type = scientific_apparatus
		visuals = fp2_armillary_sphere
		wealth = scope:wealth
		quality = scope:quality
		history = {
			type = created
			actor = root
			recipient = root
			location = root.capital_province
		}
		modifier = aov_orb_1
		save_scope_as = aov_orb
		decaying = no
	}
	set_variable = { name = aov_adv_orb value = scope:aov_orb }
}

aov_adv_orb_of_omniscience_2_effect = {
	add_magic_lifestyle_xp = 100
	if = {
		limit = { exists = var:aov_adv_orb }
		var:aov_adv_orb = { add_artifact_modifier = aov_orb_2 }
	}
}

aov_adv_orb_of_omniscience_3_effect = {

	if = {
		limit = { exists = var:aov_adv_orb }

	}
}

# One per mage per lifetime: aov_adv_enchanted_crafted stays set even if the artifact is lost
aov_adv_enchanted_store = {
	set_variable = { name = aov_adv_enchanted_artifact value = scope:newly_created_artifact }
	set_variable = { name = aov_adv_enchanted_crafted value = yes }
	set_variable = { name = aov_adv_enchanted_type value = flag:$KIND$ }
}

aov_adv_enchanted_create_weapon = {
	save_scope_as = owner
	set_artifact_rarity_illustrious = yes
	create_artifact = {
		name = aov_enchanted_weapon_name
		description = aov_enchanted_weapon_desc
		type = sword
		visuals = sword
		wealth = scope:wealth
		quality = scope:quality
		history = {
			type = created
			actor = root
			recipient = root
			location = root.capital_province
		}
		modifier = aov_enchanted_weapon_1
		save_scope_as = newly_created_artifact
		decaying = no
	}
	aov_adv_enchanted_store = { KIND = weapon }
}
