namespace = aov_hold_dig

# Random dig events at a hold with a running dig (Delve Deeper). Root = the hold's holder, scope:hold = the hold
# (province). Options add or remove months of dig progress, never below 0.
aov_hold_dig.1 = {
	type = character_event
	title = aov_hold_dig.1.t
	desc = aov_hold_dig.1.desc
	theme = stewardship
	override_background = { reference = fp3_cave }
	left_portrait = root

	trigger = {
		exists = scope:hold
		scope:hold = { aov_hold_dig_running = yes }
		scope:hold.barony.holder = root
	}

	option = {
		name = aov_hold_dig.1.a
		scope:hold = { aov_hold_add_dig_progress_effect = { MONTHS = 10 } }
	}

	option = {
		name = aov_hold_dig.1.b
		trigger = {
			gold >= 60
			scope:hold = { aov_hold_dig_running = yes }
			scope:hold.barony.holder = root
		}
		show_as_unavailable = { always = yes }
		remove_short_term_gold = 60
		scope:hold = { aov_hold_add_dig_progress_effect = { MONTHS = 16 } }
	}

	option = {
		name = aov_hold_dig.1.c
		scope:hold = { aov_hold_add_dig_progress_effect = { MONTHS = 3 } }
		add_prestige = 25
	}
}

# Breakthrough at a hold: its dig raised it one level (scope:hold, the province). Sent by
# aov_hold_check_dig_complete_effect.
aov_hold_dig.100 = {
	type = character_event
	title = aov_hold_dig.100.t
	desc = {
		desc = aov_hold_dig.100.desc
		first_valid = {
			triggered_desc = {
				trigger = { scope:aov_discovery = flag:gems }
				desc = aov_hold_dig.100.desc.gems
			}
			triggered_desc = {
				trigger = { scope:aov_discovery = flag:vein }
				desc = aov_hold_dig.100.desc.vein
			}
			desc = aov_hold_dig.100.desc.lore
		}
	}
	theme = stewardship
	override_background = { reference = fp3_cave }
	left_portrait = root

	immediate = {
		random_list = {
			1 = { save_scope_value_as = { name = aov_discovery value = flag:gems } }
			1 = { save_scope_value_as = { name = aov_discovery value = flag:vein } }
			1 = { save_scope_value_as = { name = aov_discovery value = flag:lore } }
		}
	}

	option = {
		name = aov_hold_dig.100.a
		if = {
			limit = { scope:aov_discovery = flag:gems }
			add_gold = {
				value = scope:hold.aov_hold_level
				multiply = 25
			}
		}
		else_if = {
			limit = { scope:aov_discovery = flag:vein }
			scope:hold.county = {
				add_county_modifier = {
					modifier = aov_hold_rich_vein
					years = 10
				}
			}
		}
		else = {
			add_learning_lifestyle_xp = 500
		}
	}
}
