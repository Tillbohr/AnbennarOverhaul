# Lichdom's steps, after EU4 Anbennar's Pursue Lichdom (magic_project_lichdom.<id>; ids kept). Each fires from
# aov_adv_lichdom_next_step (common/scripted_effects/aov_advancement_level_effects.txt): the first when a level starts,
# the others at a third and two thirds of its months. Faster choices add progress (aov_adv_lichdom_hasten); every cost
# is in common/script_values/aov_advancement_tuning.txt

# The Phylactery, step 1: the container (var:aov_lichdom_vessel, used when the phylactery is made)
aov_lichdom.2 = {
	type = character_event
	title = aov_lichdom.2.t
	desc = aov_lichdom.2.desc
	theme = dread
	left_portrait = root

	option = { # An amulet
		name = aov_lichdom.2.a
		remove_short_term_gold = aov_lichdom_gold_low
		set_variable = { name = aov_lichdom_vessel value = flag:amulet }
		custom_tooltip = aov_lichdom.2.a.tt
		ai_chance = { base = 50 }
	}
	option = { # A jewelled box
		name = aov_lichdom.2.b
		remove_short_term_gold = aov_lichdom_gold_mid
		set_variable = { name = aov_lichdom_vessel value = flag:box }
		custom_tooltip = aov_lichdom.2.b.tt
		aov_adv_lichdom_hasten = { SPEED = fast }
		ai_chance = { base = 50 }
	}
	option = { # EU4's back-out option
		name = aov_lichdom_back_out
		aov_adv_lichdom_back_out = yes
		ai_chance = { base = 0 }
	}
}

# The Transformation Potion, step 3: the vessel. The heir option kills the mage's primary heir (scope:aov_lichdom_heir)
aov_lichdom.7 = {
	type = character_event
	title = aov_lichdom.7.t
	desc = aov_lichdom.7.desc
	theme = dread
	left_portrait = root
	right_portrait = scope:aov_lichdom_heir

	immediate = {
		if = {
			limit = { exists = primary_heir }
			primary_heir = { save_scope_as = aov_lichdom_heir }
		}
	}

	option = { # The heart of an innocent
		name = aov_lichdom.7.a
		add_prestige = medium_prestige_loss
		add_dread = aov_lichdom_dread_low
		ai_chance = { base = 50 }
	}
	option = { # A unicorn's horn
		name = aov_lichdom.7.b
		remove_short_term_gold = aov_lichdom_gold_mid
		ai_chance = { base = 50 }
	}
	option = { # The heir
		name = aov_lichdom.7.c
		trigger = { exists = scope:aov_lichdom_heir }
		scope:aov_lichdom_heir = {
			death = {
				death_reason = death_sacrificed_to_gods
				killer = root
			}
		}
		add_dread = aov_lichdom_dread_high
		aov_adv_lichdom_hasten = { SPEED = faster }
		ai_chance = {
			base = 5
			ai_value_modifier = { ai_compassion = -0.1 }
			min = 0
		}
	}
	option = { # EU4's back-out option
		name = aov_lichdom_back_out
		aov_adv_lichdom_back_out = yes
		ai_chance = { base = 0 }
	}
}

# The Dark Ritual, step 3: the last chance to turn back (EU4's magic_project_lichdom.20)
aov_lichdom.20 = {
	type = character_event
	title = aov_lichdom.20.t
	desc = aov_lichdom.20.desc
	theme = dread
	left_portrait = root

	option = { # I will stop
		name = aov_lichdom.20.a
		if = {
			limit = { aov_adv_working_on = { KEY = lichdom } }
			add_prestige = aov_lichdom_turn_back_prestige
			aov_adv_abandon = yes
		}
		custom_tooltip = aov_lichdom.20.a.tt
		remove_variable = aov_lichdom_step
		ai_chance = {
			base = 10
			ai_value_modifier = { ai_compassion = 0.2 }
			min = 0
		}
	}
	option = { # I will not
		name = aov_lichdom.20.b
		add_dread = aov_lichdom_dread_low
		ai_chance = { base = 90 }
	}
}
