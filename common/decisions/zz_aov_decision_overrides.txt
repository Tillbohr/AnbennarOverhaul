	picture = {

		prestige = 1000

	is_shown = {
		is_landed = yes
		culture = {
			OR = {
				has_cultural_pillar = heritage_gerudian # gerudian...
				any_parent_culture_or_above = { has_cultural_pillar = heritage_gerudian } # or gerudian successor, ie pearlsedgers

			# has_title = title:k_pearlsedge
		# }

		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:restore_reaver_kingdom_of_divenhal
			}
		}
	}

	is_valid = {
		prestige_level >= very_high_prestige_level # Exalted among Men or higher
		has_trait = viking

		completely_controls_region = custom_drolas
		completely_controls = title:d_annail
		completely_controls = title:d_ibtatu
		completely_controls = title:d_nathalaire

		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
	}

	effect = {
		save_scope_as = hoarder_heir
		
		show_as_tooltip = { restore_reaver_kingdom_of_divenhal_effect = yes }
		
		trigger_event = anb_decision_major_events.0029

			name = unavailable_unique_decisions
			target = flag:restore_reaver_kingdom_of_divenhal
		}
	}

	ai_check_interval = 60
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 100
	}
