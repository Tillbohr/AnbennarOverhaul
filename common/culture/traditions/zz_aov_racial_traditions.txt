# Anbennar Overhaul: overrides three Anbennar traditions (common/culture/traditions/anb_maa_traditions.txt).
# Racial armies (docs/systems/racial-armies.md): the two dwarf traditions can never be shown or picked, and
# aov_racial_armies_game_start removes them from every culture; everything but is_shown and can_pick is unchanged.
# Racial traditions (docs/systems/racial-traditions.md): the Sundancer Legions lose their levy penalty.

# Sun elves: Anbennar's tradition without levy_size and levy_reinforcement_rate; it unlocks its two units again.
tradition_sundancer_legions = {
	category = combat

	layers = {
		0 = martial
		1 = mediterranean
		4 = laurel.dds
	}

	is_shown = {
		has_cultural_pillar = heritage_elven
	}
	can_pick = {
		OR ={
			this = culture:sun_elvish
			any_parent_culture_or_above = {
				this = culture:sun_elvish
			}
		}
	}

	parameters = {
		unlock_maa_anb_jaherkadezim = yes
		unlock_maa_anb_sundancer_legions = yes
	}
	
	character_modifier = {
		men_at_arms_recruitment_cost = -0.25
		men_at_arms_maintenance = -0.35
		men_at_arms_cap = 1
		hard_casualty_modifier = -0.15
	}
	
	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_bellicose = { is_in_list = traits }
						culture_pillar:ethos_egalitarian = { is_in_list = traits }
						culture_pillar:ethos_courtly = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_bellicose_egalitarian_or_courtly_desc
				}
			}
			
			multiply = tradition_replacement_cost_if_relevant
		}
	}
	
	ai_will_do = {
		value = 100
	}
}
