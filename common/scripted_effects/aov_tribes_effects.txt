# Anbennar Overhaul: Dwarovar tribes (docs/systems/dwarovar-tribes.md).

# Title scope, a pool title: it wears its holder's dynasty arms. Its name reads the dynasty by itself (its localization).
aov_tribe_title_sync_effect = {
	if = {
		limit = { exists = holder.dynasty }
		set_coa = holder.dynasty
	}
}

# scope:challenger beat scope:ruler in a duel: every title of the ruler and every other vassal goes to the winner. This
# runs before the ruler's death, so the realm never partitions.
aov_tribe_duel_winner_takes_all_effect = {
	create_title_and_vassal_change = {
		type = usurped
		save_scope_as = change
		add_claim_on_loss = no
	}
	scope:ruler = {
		every_held_title = {
			change_title_holder = {
				holder = scope:challenger
				change = scope:change
			}
		}
		every_vassal = {
			limit = { this != scope:challenger }
			change_liege = {
				LIEGE = scope:challenger
				CHANGE = scope:change
			}
		}
	}
	scope:challenger = {
		becomes_independent = { change = scope:change }
	}
	resolve_title_and_vassal_change = scope:change
}
