# Anbennar Overhaul: Delving Campaign event pulses. Root = the host (players only), scheduled by the
# activity's first three phases (common/activities/activity_types/aov_hold_dig.txt). Each pulse picks one
# event of its phase (events/aov_hold_dig_events.txt). Setback events weigh double under the Reckless pace
# and half under Careful: modifier factors in a random_list, which on_action random_events cannot take.

aov_hold_dig_survey_pulse = {
	effect = {
		random_list = {
			100 = {
				trigger_event = aov_hold_dig.1
			}
			100 = {
				modifier = {
					factor = 2
					involved_activity = { has_activity_option = { category = aov_dig_pace option = aov_dig_pace_reckless } }
				}
				modifier = {
					factor = 0.5
					involved_activity = { has_activity_option = { category = aov_dig_pace option = aov_dig_pace_careful } }
				}
				trigger_event = aov_hold_dig.2
			}
			100 = {
				trigger_event = aov_hold_dig.3
			}
		}
	}
}
