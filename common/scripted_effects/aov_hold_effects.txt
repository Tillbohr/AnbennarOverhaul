# Anbennar Overhaul: dwarven hold effects.

# Province scope. Starts a dig for the hold's next level (the caller checks aov_can_dig_at and pays
# aov_dig_cost). Progress counts months, from 0 up to aov_dig_months.
aov_hold_start_dig_effect = {
	if = {
		limit = {
			NOT = { aov_hold_dig_running = yes }
			aov_hold_level < 10
		}
		set_variable = {
			name = aov_dig_target_level
			value = {
				value = aov_hold_level
				add = 1
			}
		}
		set_variable = { name = aov_dig_progress value = 0 }
		if = {
			limit = { var:aov_dig_target_level = 2 }
			add_province_modifier = aov_delving_02
		}
		else_if = {
			limit = { var:aov_dig_target_level = 3 }
			add_province_modifier = aov_delving_03
		}
		else_if = {
			limit = { var:aov_dig_target_level = 4 }
			add_province_modifier = aov_delving_04
		}
		else_if = {
			limit = { var:aov_dig_target_level = 5 }
			add_province_modifier = aov_delving_05
		}
		else_if = {
			limit = { var:aov_dig_target_level = 6 }
			add_province_modifier = aov_delving_06
		}
		else_if = {
			limit = { var:aov_dig_target_level = 7 }
			add_province_modifier = aov_delving_07
		}
		else_if = {
			limit = { var:aov_dig_target_level = 8 }
			add_province_modifier = aov_delving_08
		}
		else_if = {
			limit = { var:aov_dig_target_level = 9 }
			add_province_modifier = aov_delving_09
		}
		else_if = {
			limit = { var:aov_dig_target_level = 10 }
			add_province_modifier = aov_delving_10
		}
	}
}

# Province scope. Adds $MONTHS$ months (negative for a setback) to the hold's running dig, never below 0,
# then completes the dig if it reached its length.
aov_hold_add_dig_progress_effect = {
	if = {
		limit = { aov_hold_dig_running = yes }
		if = {
			limit = { NOT = { has_variable = aov_dig_progress } }
			set_variable = { name = aov_dig_progress value = 0 }
		}
		change_variable = { name = aov_dig_progress add = $MONTHS$ }
		if = {
			limit = { var:aov_dig_progress < 0 }
			set_variable = { name = aov_dig_progress value = 0 }
		}
		aov_hold_check_dig_complete_effect = yes
	}
}

# Province scope. A dig that reached its length raises the hold to its target level (once; any surplus is
# discarded), ends, and sends the Breakthrough event (scope:hold) to the hold's holder.
aov_hold_check_dig_complete_effect = {
	if = {
		limit = {
			aov_hold_dig_running = yes
			has_variable = aov_dig_progress
			var:aov_dig_progress >= aov_dig_months
		}
		if = {
			limit = { var:aov_dig_target_level = 2 }
			add_building = dwarven_hold_02
		}
		else_if = {
			limit = { var:aov_dig_target_level = 3 }
			add_building = dwarven_hold_03
		}
		else_if = {
			limit = { var:aov_dig_target_level = 4 }
			add_building = dwarven_hold_04
		}
		else_if = {
			limit = { var:aov_dig_target_level = 5 }
			add_building = dwarven_hold_05
		}
		else_if = {
			limit = { var:aov_dig_target_level = 6 }
			add_building = dwarven_hold_06
		}
		else_if = {
			limit = { var:aov_dig_target_level = 7 }
			add_building = dwarven_hold_07
		}
		else_if = {
			limit = { var:aov_dig_target_level = 8 }
			add_building = dwarven_hold_08
		}
		else_if = {
			limit = { var:aov_dig_target_level = 9 }
			add_building = dwarven_hold_09
		}
		else_if = {
			limit = { var:aov_dig_target_level = 10 }
			add_building = dwarven_hold_10
		}
		aov_hold_end_dig_effect = yes
		save_scope_as = hold
		barony.holder = { trigger_event = aov_hold_dig.100 }
	}
}

# Province scope. Ends the hold's dig, if any: its target, progress and Delving modifier are removed.
aov_hold_end_dig_effect = {
	remove_variable = aov_dig_target_level
	remove_variable = aov_dig_progress
	if = {
		limit = { has_province_modifier = aov_delving_02 }
		remove_province_modifier = aov_delving_02
	}
	if = {
		limit = { has_province_modifier = aov_delving_03 }
		remove_province_modifier = aov_delving_03
	}
	if = {
		limit = { has_province_modifier = aov_delving_04 }
		remove_province_modifier = aov_delving_04
	}
	if = {
		limit = { has_province_modifier = aov_delving_05 }
		remove_province_modifier = aov_delving_05
	}
	if = {
		limit = { has_province_modifier = aov_delving_06 }
		remove_province_modifier = aov_delving_06
	}
	if = {
		limit = { has_province_modifier = aov_delving_07 }
		remove_province_modifier = aov_delving_07
	}
	if = {
		limit = { has_province_modifier = aov_delving_08 }
		remove_province_modifier = aov_delving_08
	}
	if = {
		limit = { has_province_modifier = aov_delving_09 }
		remove_province_modifier = aov_delving_09
	}
	if = {
		limit = { has_province_modifier = aov_delving_10 }
		remove_province_modifier = aov_delving_10
	}
}

# Province scope, a hold just conquered by a non-dwarf. Scope:previous_holder is the loser, if any.
aov_hold_ruin_effect = {
	set_variable = aov_hold_ruined
	aov_hold_end_dig_effect = yes
	save_scope_as = hold_province
	barony = { save_scope_as = hold_title }
	county = {
		add_county_modifier = {
			modifier = aov_ruined_hold
		}
	}
	if = {
		limit = {
			exists = scope:previous_holder
			scope:previous_holder = { is_alive = yes }
		}
		scope:previous_holder = { trigger_event = aov_hold_ruin.1 }
	}
	barony.holder = { trigger_event = aov_hold_ruin.1 }
}

# Province scope, root is the restorer (the holder who paid). The restoration completes in
# aov_hold_ruin.3 on whoever holds the hold then.
aov_hold_begin_restoration_effect = {
	add_province_modifier = {
		modifier = aov_restoring_hold
		years = 5
	}
	set_variable = aov_restore_pending
	set_variable = { name = aov_restore_due years = 5 }
	set_variable = { name = aov_restorer value = root }
	save_scope_as = hold_province
	barony = { save_scope_as = hold_title }
	root = { trigger_event = { id = aov_hold_ruin.3 days = 1830 } }
}
