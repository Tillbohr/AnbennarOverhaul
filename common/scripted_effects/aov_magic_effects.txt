# Anbennar Overhaul: magic schools. Character scope (the mage).
# SCHOOL = abjuration ... transmutation

# A new mage starts with full mana
aov_magic_init = {
	set_variable = { name = aov_mana value = aov_mana_max }
}

# AMOUNT: number or scope value; mana stays within 0..max
aov_mana_change = {
	if = {
		limit = { NOT = { has_variable = aov_mana } }
		set_variable = { name = aov_mana value = aov_mana_max }
	}
	change_variable = { name = aov_mana add = $AMOUNT$ }
	clamp_variable = { name = aov_mana min = 0 max = aov_mana_max }
}

# AMOUNT: number or scope value; spends mana, staying within 0..max
aov_mana_spend = {
	if = {
		limit = { NOT = { has_variable = aov_mana } }
		set_variable = { name = aov_mana value = aov_mana_max }
	}
	change_variable = { name = aov_mana subtract = $AMOUNT$ }
	clamp_variable = { name = aov_mana min = 0 max = aov_mana_max }
}

aov_study_set = {
	set_variable = { name = aov_studying value = flag:$SCHOOL$ }
}

# The Study button (aov_study_<school>_sgui): a new school opens the study event (aov_magic.2)
aov_study_choose = {
	if = {
		limit = { NOT = { aov_is_studying = { SCHOOL = $SCHOOL$ } } }
		aov_study_set = { SCHOOL = $SCHOOL$ }
		trigger_event = aov_magic.2
	}
}

# Adds progress; a crossed threshold fires the level-up event
aov_school_add_progress = {
	save_temporary_scope_value_as = { name = aov_old_level value = aov_school_$SCHOOL$_level }
	# change_variable does nothing on a variable that does not exist yet
	if = {
		limit = { NOT = { has_variable = aov_school_$SCHOOL$_progress } }
		set_variable = { name = aov_school_$SCHOOL$_progress value = 0 }
	}
	change_variable = { name = aov_school_$SCHOOL$_progress add = $AMOUNT$ }
	if = {
		limit = { aov_school_$SCHOOL$_level > scope:aov_old_level }
		save_scope_value_as = { name = aov_school value = flag:$SCHOOL$ }
		# A completed level ends the study of that school until the mage chooses to study again
		if = {
			limit = { aov_is_studying = { SCHOOL = $SCHOOL$ } }
			remove_variable = aov_studying
			save_scope_value_as = { name = aov_study_halted value = yes }
		}
		trigger_event = aov_magic.1
	}
	aov_magic_sync_mastery = yes
}

aov_magic_sync_mastery = {
	set_variable = { name = magic_mastery value = aov_magic_mastery_value }
	set_variable = { name = magic_lifestyle_total_points value = magic_lifestyle_perks }
}

### One-off spell effects (tools/data/spells.py). Caster scope.

# Scry (scope:recipient = target): reveal a secret, else intrigue experience
aov_scry_reveal = {
	save_scope_as = aov_scryer
	if = {
		limit = {
			scope:recipient = {
				any_secret = { NOT = { is_known_by = scope:aov_scryer } }
			}
		}
		scope:recipient = {
			every_secret = {
				limit = { NOT = { is_known_by = scope:aov_scryer } }
				reveal_to = scope:aov_scryer
			}
		}
	}
	else = {
		add_intrigue_lifestyle_xp = 500
	}
	# Orb of Omniscience (Divination advancement) level 2: also every secret the target knows about others
	if = {
		limit = { aov_adv_orb_of_omniscience_level >= 2 }
		scope:recipient = {
			every_known_secret = {
				limit = { NOT = { is_known_by = scope:aov_scryer } }
				reveal_to = scope:aov_scryer
			}
		}
	}
}

# Augury: one name for each threat of the coming years, shown in aov_magic.45
aov_augury = {
	save_scope_as = aov_augur
	# A claimant to one of my titles strong enough to press it
	every_held_title = {
		every_claimant = {
			limit = {
				NOT = { this = scope:aov_augur }
				max_military_strength > scope:aov_augur.max_military_strength
			}
			add_to_temporary_list = aov_augury_claimants
		}
	}
	random_in_list = {
		list = aov_augury_claimants
		save_scope_as = aov_augury_claimant
	}
	# The leader of a faction against me
	every_targeting_faction = {
		limit = { exists = faction_leader }
		faction_leader = { add_to_temporary_list = aov_augury_faction_leaders }
	}
	random_in_list = {
		list = aov_augury_faction_leaders
		save_scope_as = aov_augury_faction
	}
	# A close relative near death
	random_close_family_member = {
		limit = { health < fine_health }
		save_scope_as = aov_augury_dying
	}
	# Someone plotting against me
	every_targeting_scheme = {
		limit = { is_hostile = yes }
		scheme_owner = { add_to_temporary_list = aov_augury_plotters }
	}
	random_in_list = {
		list = aov_augury_plotters
		save_scope_as = aov_augury_plotter
	}
	trigger_event = aov_magic.45
}

# A summoned army for an army spell: at the capital, else where the caster is, bound to one of the caster's wars;
# aov_magic.10 removes it when the spell runs out. Call only when aov_spell_army_possible = yes.
# SPELL: the spell key (army name aov_<SPELL>_army_name, duration aov_spell_<SPELL>_years). TYPE, STACKS: men-at-arms.
aov_spawn_spell_army = {
	random_character_war = { save_scope_as = aov_spell_army_war }
	if = {
		limit = { exists = capital_province }
		capital_province = { save_scope_as = aov_spell_army_site }
	}
	else = {
		location = { save_scope_as = aov_spell_army_site }
	}
	spawn_army = {
		name = aov_$SPELL$_army_name
		men_at_arms = {
			type = $TYPE$
			stacks = $STACKS$
		}
		location = scope:aov_spell_army_site
		war = scope:aov_spell_army_war
		inheritable = no
		uses_supply = no
		save_scope_as = aov_spell_army
	}
	trigger_event = { id = aov_magic.10 years = aov_spell_$SPELL$_years }
}

# Nowhere to summon or no war: the spell fizzles and costs nothing
aov_spell_army_fizzle = {
	aov_mana_change = { AMOUNT = aov_spell_$SPELL$_cost }
	remove_variable = aov_spell_$SPELL$_cd
}

### AI (quarterly, aov_magic_on_actions.txt)

aov_magic_ai_quarterly = {
	if = {
		limit = { is_ai = yes }
		# Study the school that fits the best skill; once it is mastered, any unmastered school. Not while committed
		# to a magic advancement (which clears aov_studying)
		if = {
			limit = {
				NOT = { has_variable = aov_studying }
				NOT = { has_variable = aov_adv_committed }
			}
			if = {
				limit = { diplomacy >= martial diplomacy >= stewardship diplomacy >= intrigue diplomacy >= learning }
				aov_study_set = { SCHOOL = enchantment }
			}
			else_if = {
				limit = { martial >= stewardship martial >= intrigue martial >= learning }
				aov_study_set = { SCHOOL = evocation }
			}
			else_if = {
				limit = { stewardship >= intrigue stewardship >= learning }
				aov_study_set = { SCHOOL = transmutation }
			}
			else_if = {
				limit = { intrigue >= learning }
				aov_study_set = { SCHOOL = illusion }
			}
			else = {
				aov_study_set = { SCHOOL = divination }
			}
		}
		else_if = {
			limit = {
				OR = {
					AND = { aov_is_studying = { SCHOOL = abjuration } aov_school_abjuration_level >= 3 }
					AND = { aov_is_studying = { SCHOOL = conjuration } aov_school_conjuration_level >= 3 }
					AND = { aov_is_studying = { SCHOOL = divination } aov_school_divination_level >= 3 }
					AND = { aov_is_studying = { SCHOOL = enchantment } aov_school_enchantment_level >= 3 }
					AND = { aov_is_studying = { SCHOOL = evocation } aov_school_evocation_level >= 3 }
					AND = { aov_is_studying = { SCHOOL = illusion } aov_school_illusion_level >= 3 }
					AND = { aov_is_studying = { SCHOOL = necromancy } aov_school_necromancy_level >= 3 }
					AND = { aov_is_studying = { SCHOOL = transmutation } aov_school_transmutation_level >= 3 }
				}
			}
			random_list = {
				1 = {
					trigger = { aov_school_abjuration_level < 3 }
					aov_study_set = { SCHOOL = abjuration }
				}
				1 = {
					trigger = { aov_school_conjuration_level < 3 }
					aov_study_set = { SCHOOL = conjuration }
				}
				1 = {
					trigger = { aov_school_divination_level < 3 }
					aov_study_set = { SCHOOL = divination }
				}
				1 = {
					trigger = { aov_school_enchantment_level < 3 }
					aov_study_set = { SCHOOL = enchantment }
				}
				1 = {
					trigger = { aov_school_evocation_level < 3 }
					aov_study_set = { SCHOOL = evocation }
				}
				1 = {
					trigger = { aov_school_illusion_level < 3 }
					aov_study_set = { SCHOOL = illusion }
				}
				1 = {
					trigger = { aov_school_necromancy_level < 3 }
					aov_study_set = { SCHOOL = necromancy }
				}
				1 = {
					trigger = { aov_school_transmutation_level < 3 }
					aov_study_set = { SCHOOL = transmutation }
				}
			}
		}
		# Cast one spell
		if = {
			limit = { aov_magic_ai_any_castable = yes }

		}
	}
}

# Thoughtweave landed (caster scope, scope:recipient; the send option flags of aov_thoughtweave_interaction).
# Lover and rival: the caster picks the other character in aov_magic.43 from up to three candidates
aov_thoughtweave_apply = {
	save_scope_as = aov_weaver
	if = {
		limit = { scope:aov_tw_friend = yes }
		scope:recipient = {
			set_relation_friend = { target = scope:aov_weaver reason = aov_thoughtweave_reason }
		}
	}
	else_if = {
		limit = { scope:aov_tw_lover = yes }
		# The caster first, then their courtiers and the target's
		if = {
			limit = { scope:recipient = { can_set_relation_lover_trigger = { CHARACTER = scope:aov_weaver } } }
			add_to_temporary_list = aov_tw_candidates
		}
		every_courtier = {
			limit = {
				NOT = { this = scope:recipient }
				scope:recipient = { can_set_relation_lover_trigger = { CHARACTER = prev } }
			}
			add_to_temporary_list = aov_tw_candidates
		}
		scope:recipient = {
			every_courtier = {
				limit = {
					NOT = { this = scope:aov_weaver }
					scope:recipient = { can_set_relation_lover_trigger = { CHARACTER = prev } }
				}

# Dominate landed (caster scope, scope:recipient; the send option flags of aov_dominate_interaction)
aov_dominate_apply = {

		limit = {
			scope:aov_dom_surrender = yes
			aov_dominate_surrender_possible = yes
		}

			limit = {
				primary_attacker = scope:aov_dominator
				primary_defender = scope:recipient
			}

		}
		random_character_war = {
			limit = {
				primary_attacker = scope:recipient
				primary_defender = scope:aov_dominator
			}
			end_war = defender
		}
	}
	else_if = {
		limit = {
			scope:aov_dom_free = yes
			scope:recipient = { any_prisoner = { always = yes } }
		}
		scope:recipient = {
			ordered_prisoner = {
				order_by = prowess
				position = 0
				save_scope_as = aov_dom_prisoner_1
			}
			ordered_prisoner = {
				order_by = prowess
				position = 1
				save_scope_as = aov_dom_prisoner_2
			}
			ordered_prisoner = {
				order_by = prowess
				position = 2
				save_scope_as = aov_dom_prisoner_3
			}
		}
		trigger_event = aov_magic.44
	}
	else = {
		# Anbennar's own Dominate result: a strong hook, and the victim's struggle to break free
		save_scope_as = owner
		scope:recipient = { save_scope_as = target }
		dominate_success_effect = yes
	}
}
