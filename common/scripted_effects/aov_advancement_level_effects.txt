# Wards (common/modifiers/aov_advancement_modifiers.txt) on the counties the mage holds: the capital at level 1, the
# whole domain from level 2. Re-laid yearly (aov_adv_wards_yearly); they lapse after aov_adv_ward_days when the mage
# dies, loses the county or loses Magical Affinity.
aov_adv_wards_refresh = {
	every_held_county = {
		remove_county_modifier = aov_ward_1
		remove_county_modifier = aov_ward_3
		remove_county_modifier = aov_ward_capital
	}
	if = {
		limit = { aov_adv_magical_fortress_level = 1 }
		if = {
			limit = { exists = capital_county }
			capital_county = {
				add_county_modifier = { modifier = aov_ward_1 days = aov_adv_ward_days }
			}
		}
	}
	else_if = {
		limit = { aov_adv_magical_fortress_level = 2 }
		every_held_county = {
			add_county_modifier = { modifier = aov_ward_1 days = aov_adv_ward_days }
		}
		if = {
			limit = { exists = capital_county }
			capital_county = {
				add_county_modifier = { modifier = aov_ward_capital days = aov_adv_ward_days }
			}
		}
	}
	else_if = {
		limit = { aov_adv_magical_fortress_level >= 3 }
		every_held_county = {
			add_county_modifier = { modifier = aov_ward_3 days = aov_adv_ward_days }
		}
		if = {
			limit = { exists = capital_county }
			capital_county = {
				add_county_modifier = { modifier = aov_ward_capital days = aov_adv_ward_days }
			}
		}
	}
}

# Orb of Omniscience and Enchanted Artifact levels 2-3: the artifact was lost before the level completed. KEY = the
# advancement, LEVEL = the level it falls back to. The gold and mana are lost; the level must be begun again
aov_adv_artifact_level_failed = {
	set_variable = { name = aov_adv_$KEY$_lvl value = $LEVEL$ }
	save_scope_value_as = { name = aov_adv_own_event value = yes }
	trigger_event = aov_advancement.44
}

# Quarterly (aov_adv_quarterly_pulse): a mage working on a later level of the Orb or the Enchanted Artifact who no
# longer owns it stops; nothing is refunded (event aov_advancement.44)
aov_adv_artifact_check = {
	if = {
		limit = {
			aov_adv_working_on = { KEY = enchanted_artifact }
			aov_adv_enchanted_artifact_level >= 1
			aov_adv_owns_enchanted_artifact = no
		}
		save_scope_value_as = { name = aov_adv value = flag:enchanted_artifact }
		aov_adv_abandon = yes
		trigger_event = aov_advancement.44
	}
	else_if = {
		limit = {
			aov_adv_working_on = { KEY = orb_of_omniscience }
			aov_adv_orb_of_omniscience_level >= 1
			aov_adv_owns_orb = no
		}
		save_scope_value_as = { name = aov_adv value = flag:orb_of_omniscience }
		aov_adv_abandon = yes
		trigger_event = aov_advancement.44
	}
}

# Recruit Battlemage Commander (common/decisions/aov_battlemage_decisions.txt): a mage commander, better with the
# academy's tier; tier 3 recruits are trained to command (aov_battlemage_trained)
aov_recruit_battlemage = {
	if = {
		limit = { aov_battlemage_academy_tier_at_least = { TIER = 3 } }
		create_character = {
			template = aov_battlemage_template_3
			location = root.capital_province
			culture = root.culture
			faith = root.faith
			save_scope_as = aov_battlemage
		}
		scope:aov_battlemage = {
			add_character_modifier = { modifier = aov_battlemage_trained }
			set_variable = { name = aov_school_evocation_progress value = aov_battlemage_evocation_3 }
		}
	}
	else_if = {
		limit = { aov_battlemage_academy_tier_at_least = { TIER = 2 } }
		create_character = {
			template = aov_battlemage_template_2
			location = root.capital_province
			culture = root.culture
			faith = root.faith
			save_scope_as = aov_battlemage
		}
		scope:aov_battlemage = {
			set_variable = { name = aov_school_evocation_progress value = aov_battlemage_evocation_2 }
		}
	}
	else = {
		create_character = {
			template = aov_battlemage_template_1
			location = root.capital_province
			culture = root.culture
			faith = root.faith
			save_scope_as = aov_battlemage
		}
		scope:aov_battlemage = {
			set_variable = { name = aov_school_evocation_progress value = aov_battlemage_evocation_1 }
		}
	}
	scope:aov_battlemage = {
		add_trait = magical_affinity_1
		aov_magic_init = yes
		aov_magic_sync_mastery = yes
	}
	add_courtier = scope:aov_battlemage
	trigger_event = aov_advancement.70
}

# Necromancy: Lichdom
##########
# EU4 Anbennar's Pursue Lichdom: each level is three steps, each a choice event (events/aov_lichdom_events.txt, ids
# after EU4's magic_project_lichdom events). The first fires when the level starts (aov_adv_lichdom_on_start), the
# second and third once a third and two thirds of the level's months have passed (aov_adv_lichdom_steps, quarterly).
# var:aov_lichdom_step = steps fired in the level in progress

# The Phylactery (var:aov_adv_phylactery): an amulet or a box (var:aov_lichdom_vessel, step aov_lichdom.2). Losing it
# before the mage becomes a Lich starts Lichdom over (aov_adv_lichdom_check)
aov_adv_lichdom_1_effect = {
	add_magic_lifestyle_xp = 100
	remove_variable = aov_lichdom_step
	save_scope_as = owner
	set_artifact_rarity_famed = yes
	if = {
		limit = {
			has_variable = aov_lichdom_vessel
			var:aov_lichdom_vessel = flag:box
		}
		create_artifact = {
			name = aov_phylactery_box_name
			description = aov_phylactery_box_desc
			type = pedestal
			visuals = small_box
			wealth = scope:wealth
			quality = scope:quality
			history = {
				type = created
				actor = root
				recipient = root
				location = root.location
			}
			modifier = aov_phylactery
			save_scope_as = aov_phylactery
			decaying = no
		}
	}
	else = {
		create_artifact = {
			name = aov_phylactery_name
			description = aov_phylactery_desc
			type = miscellaneous
			visuals = necklace
			wealth = scope:wealth
			quality = scope:quality
			history = {
				type = created
				actor = root
				recipient = root
				location = root.location
			}
			modifier = aov_phylactery
			save_scope_as = aov_phylactery
			decaying = no
		}
	}
	set_variable = { name = aov_adv_phylactery value = scope:aov_phylactery }
}

# The Transformation Potion: kept until it is drunk in the Dark Ritual
aov_adv_lichdom_2_effect = {
	add_magic_lifestyle_xp = 100
	remove_variable = aov_lichdom_step
	if = {
		limit = { aov_adv_owns_phylactery = no }
		save_scope_value_as = { name = aov_adv_own_event value = yes }
		aov_adv_lichdom_check = yes
	}
	else = {
		add_character_modifier = { modifier = aov_lichdom_potion }
		aov_dark_magic_roll = yes
	}
}

# The Dark Ritual: the mage drinks the potion, dies and rises a Lich (event aov_lichdom.10). Without the phylactery
# Lichdom starts over (aov_adv_lichdom_check, event aov_advancement.41); if the lich trait cannot be added the ritual
# fails (event aov_advancement.43)
aov_adv_lichdom_3_effect = {
	add_magic_lifestyle_xp = 100
	remove_variable = aov_lichdom_step
	save_scope_value_as = { name = aov_adv_own_event value = yes }
	if = {
		limit = { aov_adv_owns_phylactery = no }
		aov_adv_lichdom_check = yes
	}
	else_if = {
		limit = { aov_adv_can_become_lich = no }
		set_variable = { name = aov_adv_lichdom_lvl value = 2 }
		trigger_event = aov_advancement.43
	}
	else = {
		trigger_event = aov_lichdom.10
	}
}

# Drinking the potion (event aov_lichdom.10): the mage becomes a Lich and is known at once as an Evil Mage, whatever
# their faith thinks of magic. The skeleton portrait follows the trait (gfx/portraits/portrait_modifiers/aov_lich.txt)
aov_adv_lich_ascend = {
	add_trait = lich
	aov_race_relations_update_effect = yes # monstrous from now on (docs/systems/race-relations.md)
	aov_lich_maa_on_becoming_lich = yes
	if = {
		limit = { has_character_modifier = aov_lichdom_potion }
		remove_character_modifier = aov_lichdom_potion
	}
	if = {
		limit = { NOT = { has_trait = forbidden_magic_practitioner } }
		gain_forbidden_magic_trait_effect = yes
	}
}

# Level start (aov_adv_lichdom_start, generated): the level's first step
aov_adv_lichdom_on_start = {
	set_variable = { name = aov_lichdom_step value = 0 }
	aov_adv_lichdom_next_step = yes
}

# Quarterly (aov_adv_quarterly_pulse), after the progress tick: the level's next step once its share of months is done
aov_adv_lichdom_steps = {
	if = {
		limit = {
			aov_adv_working_on = { KEY = lichdom }
			has_variable = aov_lichdom_step
			var:aov_lichdom_step < 3
			var:aov_adv_progress >= aov_lichdom_next_step_at
		}
		aov_adv_lichdom_next_step = yes
	}
}

# Fires the step event: level 1 is EU4's steps 2-4 (the phylactery), level 2 steps 5-7 (the potion), level 3 steps 8,
# 9 and 20 (the ritual's preparation, cleansing and the last chance to turn back)
aov_adv_lichdom_next_step = {
	change_variable = { name = aov_lichdom_step add = 1 }
	if = {
		limit = { var:aov_adv_working_level = 1 }
		if = {
			limit = { var:aov_lichdom_step = 1 }
			trigger_event = aov_lichdom.2
		}
		else_if = {
			limit = { var:aov_lichdom_step = 2 }
			trigger_event = aov_lichdom.3
		}
		else = {
			trigger_event = aov_lichdom.4
		}
	}
	else_if = {
		limit = { var:aov_adv_working_level = 2 }
		if = {
			limit = { var:aov_lichdom_step = 1 }
			trigger_event = aov_lichdom.5
		}
		else_if = {
			limit = { var:aov_lichdom_step = 2 }
			trigger_event = aov_lichdom.6
		}
		else = {
			trigger_event = aov_lichdom.7
		}
	}
	else = {
		if = {
			limit = { var:aov_lichdom_step = 1 }
			trigger_event = aov_lichdom.8
		}
		else_if = {
			limit = { var:aov_lichdom_step = 2 }
			trigger_event = aov_lichdom.9
		}
		else = {
			trigger_event = aov_lichdom.20
		}
	}
}

# A faster choice in a step: SPEED = fast or faster (aov_lichdom_months_<SPEED>). Progress is capped so each step left
# still gets its own quarter before the level completes (aov_lichdom_progress_cap)
aov_adv_lichdom_hasten = {
	custom_tooltip = aov_lichdom_hasten_$SPEED$_tt
	hidden_effect = {
		if = {
			limit = {
				aov_adv_working_on = { KEY = lichdom }
				has_variable = aov_adv_progress
				has_variable = aov_lichdom_step
			}
			change_variable = { name = aov_adv_progress add = aov_lichdom_months_$SPEED$ }
			clamp_variable = { name = aov_adv_progress max = aov_lichdom_progress_cap }
		}
	}
}

# "Let us put an end to this madness!": the level in progress is abandoned; it costs more prestige the deeper the mage
# has gone (EU4's back-out option)
aov_adv_lichdom_back_out = {
	custom_tooltip = aov_lichdom_back_out_tt
	if = {
		limit = { aov_adv_working_on = { KEY = lichdom } }
		add_prestige = aov_lichdom_back_out_prestige
		aov_adv_abandon = yes
	}
	remove_variable = aov_lichdom_step
}

# Quarterly (aov_adv_quarterly_pulse) and at levels 2-3: a mage who has crafted the phylactery but lost it before
# becoming a Lich starts Lichdom over; the potion, bound to the phylactery, is lost with it
aov_adv_lichdom_check = {
	if = {
		limit = {
			NOT = { has_trait = lich }
			aov_adv_lichdom_level >= 1
			aov_adv_owns_phylactery = no
		}
		set_variable = { name = aov_adv_lichdom_lvl value = 0 }
		if = {
			limit = { has_variable = aov_adv_phylactery }
			remove_variable = aov_adv_phylactery
		}
		if = {
			limit = { has_character_modifier = aov_lichdom_potion }
			remove_character_modifier = aov_lichdom_potion
		}
		if = {
			limit = { aov_adv_working_on = { KEY = lichdom } }
			aov_adv_abandon = yes
		}
		trigger_event = aov_advancement.41
	}
}

	trigger_event = aov_advancement.30

# A homunculus courtier at the mage's location (var:aov_adv_homunculus): sterile, cannot inherit, devoted to its maker
aov_adv_homunculus_create = {
	create_character = {
		template = aov_homunculus_template
		location = root.location
		culture = root.culture
		faith = root.faith
		save_scope_as = aov_homunculus
	}
	scope:aov_homunculus = {
		add_trait = aov_homunculus
		add_opinion = {
			target = root
			modifier = aov_homunculus_maker_opinion
		}
	}
	add_courtier = scope:aov_homunculus
	set_variable = { name = aov_adv_homunculus value = scope:aov_homunculus }
}
