# Anbennar Overhaul: magic schools. Character scope.

# A mage: any Magical Affinity (Anbennar)
aov_is_mage = {
	has_magical_affinity = yes
}

# SCHOOL = abjuration ... transmutation, LEVEL = 0-3
aov_school_level_at_least = {
	aov_school_$SCHOOL$_level >= $LEVEL$
}

aov_is_studying = {
	has_variable = aov_studying
	var:aov_studying = flag:$SCHOOL$
}

# Necromancy is dark magic where the faith makes magic illegal or shunned
aov_dark_magic_risky = {
	faith = {
		OR = {
			has_doctrine_parameter = witchcraft_illegal
			has_doctrine_parameter = witchcraft_shunned
		}
	}
}

# Recipient of a targeted spell: not protected by Field of Forbiddance
aov_spell_target_allowed = {
	custom_tooltip = {
		text = AOV_SPELL_REQ_NOT_FORBIDDEN
		NOT = { has_character_modifier = aov_spell_field_of_forbiddance }
	}
}

# Has one of the magic lifestyle's focuses (common/focuses/zz_aov_magic_focuses.txt), like vanilla's has_focus_learning
aov_has_focus_magic = {
	OR = {
		has_focus = magic_arcane_study_focus
		has_focus = magic_duelist_focus
		has_focus = magic_mindweaving_focus
	}
}

# An army spell has somewhere to summon and a war to fight (aov_spawn_spell_army)
aov_spell_army_possible = {
	OR = {
		exists = capital_province
		exists = location
	}
	any_character_war = { always = yes }
}

# AI: a worthy target for a hostile spell (interaction scopes)
aov_spell_ai_hostile_target = {
	OR = {
		scope:actor = { has_relation_rival = scope:recipient }
		scope:actor = { is_at_war_with = scope:recipient }
	}
}

# Dominate: the caster and the target lead the two sides of a war (interaction scopes)
aov_dominate_surrender_possible = {
	scope:actor = {
		any_character_war = {
			OR = {
				AND = {
					primary_attacker = scope:actor
					primary_defender = scope:recipient
				}
				AND = {
					primary_attacker = scope:recipient
					primary_defender = scope:actor
				}
			}
		}
	}
}

# Fireball: a ruler at war with the caster, or a knight or army commander of one (interaction scopes)
aov_spell_target_war_foe = {
	custom_tooltip = {
		text = AOV_SPELL_REQ_WAR_FOE
		OR = {
			scope:actor = { is_at_war_with = scope:recipient }
			scope:recipient = {
				OR = {
					is_knight = yes
					is_commanding_army = yes
				}
				exists = liege
				liege = { is_at_war_with = scope:actor }
			}
		}
	}
}

# Winter's Grip, Earthquake: a ruler at war with the caster who has a capital (interaction scopes)
aov_spell_target_war_capital = {
	custom_tooltip = {
		text = AOV_SPELL_REQ_WAR_CAPITAL
		scope:actor = { is_at_war_with = scope:recipient }
		scope:recipient = { exists = capital_province }
	}
}

# Grand Deception: the target holds a county (interaction scopes)
aov_spell_target_has_county = {
	custom_tooltip = {
		text = AOV_SPELL_REQ_HAS_COUNTY
		scope:recipient = { any_held_county = { count >= 1 } }
	}
}

# Plague Cloud: the target has a capital for the outbreak (interaction scopes)
aov_spell_target_has_capital = {
	custom_tooltip = {
		text = AOV_SPELL_REQ_HAS_CAPITAL
		scope:recipient = { exists = capital_province }
	}
}
