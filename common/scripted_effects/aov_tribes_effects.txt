# Anbennar Overhaul: Dwarovar tribes (docs/systems/dwarovar-tribes.md).

# Title scope, a pool title: it wears its holder's dynasty arms. Its name reads the dynasty by itself (its localization).
aov_tribe_title_sync_effect = {
	if = {
		limit = { exists = holder.dynasty }
		set_coa = holder.dynasty
	}
}

# scope:challenger beat scope:ruler in a duel: the ruler's top title and their other vassals go to the winner, as in
# vanilla's tribal challenge. The ruler becomes the winner's vassal with their other titles, unless they hold another
# title of the same rank. Their death comes after this, so those titles pass to their heir.
aov_tribe_duel_winner_takes_title_effect = {
	scope:ruler = {
		primary_title = { save_scope_as = title_being_fought_over }
		if = {
			limit = {
				any_held_title = {
					count = 0
					this != scope:title_being_fought_over
					tier = scope:title_being_fought_over.tier
				}
			}
			add_to_temporary_list = transferred_vassals
		}
		every_vassal = {
			limit = { this != scope:challenger }
			add_to_temporary_list = transferred_vassals
		}
	}
	create_title_and_vassal_change = {
		type = usurped
		save_scope_as = change
		add_claim_on_loss = no
	}
	scope:title_being_fought_over = {
		change_title_holder = {
			holder = scope:challenger
			change = scope:change
		}
	}
	every_in_list = {
		list = transferred_vassals
		change_liege = {
			LIEGE = scope:challenger
			CHANGE = scope:change
		}
	}
	scope:challenger = {
		becomes_independent = { change = scope:change }
	}
	resolve_title_and_vassal_change = scope:change
}
