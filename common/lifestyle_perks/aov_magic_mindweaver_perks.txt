# Silver Tongue
aov_silver_tongue_perk = {
	lifestyle = magic_lifestyle
	tree = aov_mindweaver
	position = { 0 0 }
	icon = node_magic

	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				OR = {
					aov_is_studying = { SCHOOL = enchantment }
					aov_is_studying = { SCHOOL = illusion }
					aov_is_studying = { SCHOOL = necromancy }
				}
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = magic_mindweaving_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = aov_veil_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		diplomacy = 1
	}

	effect = {
		custom_description_no_bullet = { text = aov_silver_tongue_perk_effect }
		aov_magic_sync_mastery = yes
	}
}
