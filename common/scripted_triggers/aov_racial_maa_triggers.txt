# Anbennar Overhaul: culture lineage triggers for racial men-at-arms (culture scope).
# A culture belongs to a lineage when it is the root culture or descends from it (divergence or hybridization),
# so splinter cultures keep their units. See docs/systems/racial-armies.md.

# Sun elves: sun_elvish and its descendants.
aov_sun_elven_lineage_trigger = {
	OR = {
		this = culture:sun_elvish
		any_parent_culture_or_above = { this = culture:sun_elvish }
	}
}

# Wood elves: wood_elvish and its descendants.
aov_wood_elven_lineage_trigger = {
	OR = {
		this = culture:wood_elvish
		any_parent_culture_or_above = { this = culture:wood_elvish }
	}
}

# Moon elves: every other elven culture. Wood elves descend from moon_elvish, so both other lineages are
# excluded explicitly; a sun/moon hybrid takes the sun roster.
aov_moon_elven_lineage_trigger = {
	has_cultural_pillar = heritage_elven
	aov_sun_elven_lineage_trigger = no
	aov_wood_elven_lineage_trigger = no
}
